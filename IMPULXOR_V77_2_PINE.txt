// MTF — V73: UNA SOLA REQUEST POR TEMPORALIDAD
// Antes: 4 requests en f_dir + 1 en f_rsi por cada TF (35 llamadas).

f_mtf(tf) =>
    [c, e9, e21, rr] = request.security(syminfo.tickerid, tf, [close, ta.ema(close, emaFastLen), ta.ema(close, emaMidLen), ta.rsi(close, rsiLen)], barmerge.gaps_off, barmerge.lookahead_off)
    d = c > e9 and e9 > e21 and rr > 50 ? 1 : c < e9 and e9 < e21 and rr < 50 ? -1 : 0
    [d, rr]

[d1, r1]     = f_mtf("1")
[d5, r5]     = f_mtf("5")
[d15, r15]   = f_mtf("15")
[d30, r30]   = f_mtf("30")
[d45, r45]   = f_mtf("45")
[d60, r60]   = f_mtf("60")
[d120, r120] = f_mtf("120")
[d240, r240] = f_mtf("240")
[dD, rD]     = f_mtf("D")

// Fuerza visible por temporalidad. Score direccional 0-100, no probabilidad.
// V73: devuelve 0 cuando el RSI del marco aún no existe (historial corto de M1).
f_tfStrengthV67(d, rr) =>
    na(rr) ? 0 :
     d == 0 ? math.round(math.min(49.0, math.abs(rr - 50.0) * 1.55)) :
     math.round(math.min(100.0, 55.0 + math.abs(rr - 50.0) * 1.80))

f_uiGroupV76(_s) =>
    // separador de miles en la parte entera de un número ya convertido a texto ("4287.00" -> "4,287.00")
    neg = str.startswith(_s, "-")
    bodyTxt = neg ? str.substring(_s, 1) : _s
    dot = str.pos(bodyTxt, ".")
    intPart = na(dot) ? bodyTxt : str.substring(bodyTxt, 0, dot)
    decPart = na(dot) ? "" : str.substring(bodyTxt, dot)
    n = str.length(intPart)
    out = ""
    if n > 0
        for i = 0 to n - 1
            out += str.substring(intPart, i, i + 1)
            if (n - 1 - i) % 3 == 0 and i < n - 1
                out += ","
    (neg ? "-" : "") + out + decPart
