// MTF — V73: UNA SOLA REQUEST POR TEMPORALIDAD
// Antes: 4 requests en f_dir + 1 en f_rsi por cada TF (35 llamadas).

f_mtf(tf) =>
    [c, e9, e21, rr] = request.security(syminfo.tickerid, tf, [close, ta.ema(close, emaFastLen), ta.ema(close, emaMidLen), ta.rsi(close, rsiLen)], barmerge.gaps_off, barmerge.lookahead_off)
    d = c > e9 and e9 > e21 and rr > 50 ? 1 : c < e9 and e9 < e21 and rr < 50 ? -1 : 0
    [d, rr]

[d5, r5]     = f_mtf("5")
[d15, r15]   = f_mtf("15")
[d30, r30]   = f_mtf("30")
[d45, r45]   = f_mtf("45")
[d60, r60]   = f_mtf("60")
[d120, r120] = f_mtf("120")
[d240, r240] = f_mtf("240")
[dD, rD]     = f_mtf("D")

    na(rr) ? 0 :
     d == 0 ? math.round(math.min(49.0, math.abs(rr - 50.0) * 1.55)) :
     math.round(math.min(100.0, 55.0 + math.abs(rr - 50.0) * 1.80))

seqAbsorbSell = math.sum(upperWick > bodySafe * 0.50 and close < high - rng * 0.45 ? 1.0 : 0.0, 3) >= 2

trapBuy  = sweepDown and close > open and (absorbBuy or seqAbsorbBuy or pocDefendedBuy)

sellContinuation = sellTrendAlive and (sellPullbackDying or absorbSell or seqAbsorbSell or microBosSell or pocDefendedSell) and diMinus >= diPlus
