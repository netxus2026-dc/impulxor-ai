// Helpers de texto seguros ante na (evita "NaN" en paneles y alertas).
f_px(_v) => na(_v) ? "-" : str.tostring(_v, format.mintick)
f_range(_a, _b) => na(_a) or na(_b) ? "-" : str.tostring(_a, format.mintick) + " - " + str.tostring(_b, format.mintick)

//━━━━━━━━━━━━━━━━━━
// MTF — V73: UNA SOLA REQUEST POR TEMPORALIDAD
// Antes: 4 requests en f_dir + 1 en f_rsi por cada TF (35 llamadas).
// Ahora: 1 tupla por TF (7 llamadas). Misma lógica de dirección.
//━━━━━━━━━━━━━━━━━━
f_mtf(tf) =>
    [c, e9, e21, rr] = request.security(syminfo.tickerid, tf, [close, ta.ema(close, emaFastLen), ta.ema(close, emaMidLen), ta.rsi(close, rsiLen)], barmerge.gaps_off, barmerge.lookahead_off)
    d = c > e9 and e9 > e21 and rr > 50 ? 1 : c < e9 and e9 < e21 and rr < 50 ? -1 : 0
    [d, rr]

[d1, r1]     = f_mtf("1")
[d5, r5]     = f_mtf("5")
[d15, r15]   = f_mtf("15")
[d30, r30]   = f_mtf("30")
[d45, r45]   = f_mtf("45")
[d60, r60]   = f_mtf("60")
[d120, r120] = f_mtf("120")
[d240, r240] = f_mtf("240")
[dD, rD]     = f_mtf("D")

// H3 institucional: incluye rango de 36 velas para el motor madre.
[h3Close, h3E9, h3E21, r180, h3HighRaw, h3LowRaw] = request.security(syminfo.tickerid, "180", [close, ta.ema(close, emaFastLen), ta.ema(close, emaMidLen), ta.rsi(close, rsiLen), ta.highest(high[1], 36), ta.lowest(low[1], 36)], barmerge.gaps_off, barmerge.lookahead_off)
d180 = h3Close > h3E9 and h3E9 > h3E21 and r180 > 50 ? 1 : h3Close < h3E9 and h3E9 < h3E21 and r180 < 50 ? -1 : 0

// Fuerza visible por temporalidad. Score direccional 0-100, no probabilidad.
// V73: devuelve 0 cuando el RSI del marco aún no existe (historial corto de M1).
f_tfStrengthV67(d, rr) =>
    na(rr) ? 0 :
     d == 0 ? math.round(math.min(49.0, math.abs(rr - 50.0) * 1.55)) :
     math.round(math.min(100.0, 55.0 + math.abs(rr - 50.0) * 1.80))

tfDirTxt(d) => d == 1 ? "BUY" : d == -1 ? "SELL" : "NEUTRO"

mtfBuy  = (d1 == 1 ? 6 : 0) + (d5 == 1 ? 16 : 0) + (d15 == 1 ? 18 : 0) + (d30 == 1 ? 10 : 0) + (d45 == 1 ? 8 : 0) + (d60 == 1 ? 10 : 0) + (d120 == 1 ? 8 : 0) + (d180 == 1 ? 8 : 0) + (d240 == 1 ? 8 : 0) + (dD == 1 ? 8 : 0)
mtfSell = (d1 == -1 ? 6 : 0) + (d5 == -1 ? 16 : 0) + (d15 == -1 ? 18 : 0) + (d30 == -1 ? 10 : 0) + (d45 == -1 ? 8 : 0) + (d60 == -1 ? 10 : 0) + (d120 == -1 ? 8 : 0) + (d180 == -1 ? 8 : 0) + (d240 == -1 ? 8 : 0) + (dD == -1 ? 8 : 0)

//━━━━━━━━━━━━━━━━━━
// LIQUIDEZ / ZONAS
//━━━━━━━━━━━━━━━━━━
[dayHigh, dayLow, prevDayHigh, prevDayLow] = request.security(syminfo.tickerid, "D", [high, low, high[1], low[1]], barmerge.gaps_off, barmerge.lookahead_off)
dayRange = math.max(dayHigh - dayLow, syminfo.mintick)
fib31 = dayLow + dayRange * 0.31
fib50 = dayLow + dayRange * 0.50
fib68 = dayLow + dayRange * 0.68

liqHigh = ta.highest(high[1], liqLookback)
liqLow  = ta.lowest(low[1], liqLookback)
liqHighFast = ta.highest(high[1], liqSwingFast)
liqLowFast  = ta.lowest(low[1], liqSwingFast)
liqHighSlow = ta.highest(high[1], liqSwingSlow)
liqLowSlow  = ta.lowest(low[1], liqSwingSlow)

upperLiquidity = math.max(liqHigh, math.max(liqHighFast, liqHighSlow))
lowerLiquidity = math.min(liqLow, math.min(liqLowFast, liqLowSlow))

buyBaseLow  = math.min(lowerLiquidity - atr * zoneAtr, ema21 - atr * 0.35)
buyBaseHigh = math.max(lowerLiquidity + atr * zoneAtr, ema21 + atr * 0.35)
buyZoneFrom = math.min(buyBaseLow, fib31)
buyZoneTo   = math.max(buyBaseHigh, fib50)

sellBaseLow  = math.min(upperLiquidity - atr * zoneAtr, ema21 - atr * 0.35)
sellBaseHigh = math.max(upperLiquidity + atr * zoneAtr, ema21 + atr * 0.35)
sellZoneFrom = math.min(sellBaseLow, fib50)
sellZoneTo   = math.max(sellBaseHigh, fib68)

inBuyZone  = low <= buyZoneTo and high >= buyZoneFrom
inSellZone = high >= sellZoneFrom and low <= sellZoneTo

//━━━━━━━━━━━━━━━━━━
// CANAL / SOP RES
//━━━━━━━━━━━━━━━━━━
channelLen = 34
chHigh = ta.highest(high, channelLen)
chLow  = ta.lowest(low, channelLen)
chRange = math.max(chHigh - chLow, syminfo.mintick)
inChannel = chRange <= atr * 8.0

supportTouch = low <= chLow + atr * 0.30 or inBuyZone
resistTouch  = high >= chHigh - atr * 0.30 or inSellZone

supportStrength = math.round(math.min(100.0, math.max(0.0, 35.0 + (supportTouch ? 25.0 : 0.0) + (lowerWick > bodySafe ? 15.0 : 0.0) + (d15 == 1 ? 15.0 : 0.0) + (diPlus > diMinus ? 10.0 : 0.0))))
resistStrength  = math.round(math.min(100.0, math.max(0.0, 35.0 + (resistTouch ? 25.0 : 0.0) + (upperWick > bodySafe ? 15.0 : 0.0) + (d15 == -1 ? 15.0 : 0.0) + (diMinus > diPlus ? 10.0 : 0.0))))

nearSupport = math.abs(close - chLow) <= atr * nearAtr or math.abs(close - buyZoneTo) <= atr * nearAtr
nearResist  = math.abs(close - chHigh) <= atr * nearAtr or math.abs(close - sellZoneFrom) <= atr * nearAtr

//━━━━━━━━━━━━━━━━━━
// FLOW
//━━━━━━━━━━━━━━━━━━
volSafe = nz(volume, 1.0)
volBase20 = ta.sma(volume, 20)   // V73: fuera del panel para que ta.sma se calcule en cada vela.
clv = ((close - low) - (high - close)) / rng
flowRaw = clv * volSafe
flowFast = ta.ema(flowRaw, 3)
flowSlow = ta.ema(flowRaw, 13)
flowAbsMa = math.max(ta.sma(math.abs(flowRaw), 20), 1.0)

buyPressure  = flowFast > flowSlow and flowFast > 0
sellPressure = flowFast < flowSlow and flowFast < 0

deltaSynthetic = math.round(flowFast / flowAbsMa * 12.0)
deltaAccelerationBuy  = flowFast > flowFast[1] and buyPressure
deltaAccelerationSell = flowFast < flowFast[1] and sellPressure
pocDefendedBuy  = close > ema9 and lowerWick > bodySafe * 0.45 and buyPressure
pocDefendedSell = close < ema9 and upperWick > bodySafe * 0.45 and sellPressure

//━━━━━━━━━━━━━━━━━━
// MICROESTRUCTURA
//━━━━━━━━━━━━━━━━━━
compression = ta.sma(rng, 5) < ta.sma(rng, 20) * 0.78 and ta.highest(rng, 5) < atr * 1.10
adxWake = adx > adx[1] and adx[1] >= adx[2] and adx > 14

sweepDown = low < lowerLiquidity - atr * 0.04 and close > lowerLiquidity
sweepUp   = high > upperLiquidity + atr * 0.04 and close < upperLiquidity

microBosBuy  = close > ta.highest(high[1], 3)
microBosSell = close < ta.lowest(low[1], 3)

absorbBuy  = inBuyZone and (lowerWick > bodySafe * 0.70 or sweepDown) and close > low + rng * 0.55 and (buyPressure or diPlus >= diMinus)
absorbSell = inSellZone and (upperWick > bodySafe * 0.70 or sweepUp) and close < high - rng * 0.55 and (sellPressure or diMinus >= diPlus)

seqAbsorbBuy  = math.sum(lowerWick > bodySafe * 0.50 and close > low + rng * 0.45 ? 1.0 : 0.0, 3) >= 2
seqAbsorbSell = math.sum(upperWick > bodySafe * 0.50 and close < high - rng * 0.45 ? 1.0 : 0.0, 3) >= 2

trapBuy  = sweepDown and close > open and (absorbBuy or seqAbsorbBuy or pocDefendedBuy)
trapSell = sweepUp and close < open and (absorbSell or seqAbsorbSell or pocDefendedSell)

    if array.size(heatBoxesV63) > 0
        for j = array.size(heatBoxesV63) - 1 to 0
            box.delete(array.get(heatBoxesV63, j))
        array.clear(heatBoxesV63)

    heatRange = math.max(heatHiV63 - heatLoV63, syminfo.mintick)
    heatStep = heatRange / heatBinsV63
    vols = array.new_float(heatBinsV63, 0.0)
    buys = array.new_float(heatBinsV63, 0.0)

    for i = 0 to heatLookbackV63 - 1
        mid = (high[i] + low[i]) * 0.5
        idx = int(math.floor((mid - heatLoV63) / heatStep))
        idx := math.max(0, math.min(heatBinsV63 - 1, idx))
        v = nz(volume[i], 0.0)
        array.set(vols, idx, array.get(vols, idx) + v)
        buyShare = close[i] >= open[i] ? 0.68 : 0.32
        array.set(buys, idx, array.get(buys, idx) + v * buyShare)

    maxV = 0.0
    for b = 0 to heatBinsV63 - 1
        maxV := math.max(maxV, array.get(vols, b))

    for b = 0 to heatBinsV63 - 1
        v = array.get(vols, b)
        if v > 0 and maxV > 0
            y0 = heatLoV63 + b * heatStep
            y1 = y0 + heatStep * 0.88
            ratio = array.get(buys, b) / v
            strength = v / maxV
            w = 1 + int(math.round(strength * 9))
            col = ratio >= 0.56 ? color.new(v63Cyan, 35) : ratio <= 0.44 ? color.new(v63Red, 35) : color.new(v63Gray, 55)
            bx = box.new(bar_index + 3, y1, bar_index + 3 + w, y0, xloc=xloc.bar_index, bgcolor=col, border_color=color.new(col, 0), border_width=1)
            array.push(heatBoxesV63, bx)
